cmake_minimum_required(VERSION 3.20)
project(limo_occt_storage_check LANGUAGES CXX)
add_executable(limo_occt_storage_check math-double-tab-check.cxx)
target_compile_features(limo_occt_storage_check PRIVATE cxx_std_17)
target_include_directories(limo_occt_storage_check PRIVATE "${OCCT_INCLUDE}")
find_library(OCCT_TKMATH NAMES TKMath PATHS "${OCCT_LIB}" NO_DEFAULT_PATH REQUIRED)
find_library(OCCT_TKERNEL NAMES TKernel PATHS "${OCCT_LIB}" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(limo_occt_storage_check PRIVATE "${OCCT_TKMATH}" "${OCCT_TKERNEL}")
set_target_properties(limo_occt_storage_check PROPERTIES BUILD_RPATH "${OCCT_LIB}")
